
import fs from "node:fs/promises";
import path from "node:path";

import {
  AlignmentType,
  BorderStyle,
  Document,
  Footer,
  Header,
  ImageRun,
  PageNumber,
  Paragraph,
  Packer,
  Table,
  TableCell,
  TableRow,
  TextRun,
  WidthType,
} from "docx";

const FONT = "Times New Roman";
const BLACK = "000000";
const BLUE = "1F3864";
const GRAY = "666666";

function text(value, options = {}) {
  return new TextRun({
    text: String(value ?? ""),
    font: FONT,
    size: options.size ?? 22,
    bold: options.bold ?? false,
    italics: options.italics ?? false,
    color: options.color ?? BLACK,
  });
}

function paragraph(value, options = {}) {
  return new Paragraph({
    alignment: options.alignment ?? AlignmentType.JUSTIFIED,
    spacing: {
      before: options.before ?? 0,
      after: options.after ?? 120,
      line: 300,
    },
    indent: options.indent ? { firstLine: 360 } : undefined,
    children: [
      text(value, {
        size: options.size ?? 22,
        bold: options.bold,
        color: options.color,
      }),
    ],
  });
}

function heading(value) {
  return new Paragraph({
    keepNext: true,
    spacing: { before: 220, after: 100 },
    children: [
      text(value, {
        size: 24,
        bold: true,
        color: BLUE,
      }),
    ],
    border: {
      bottom: {
        style: BorderStyle.SINGLE,
        size: 6,
        color: BLUE,
        space: 4,
      },
    },
  });
}

function borders(color = BLACK, size = 6) {
  const edge = {
    style: BorderStyle.SINGLE,
    size,
    color,
  };

  return {
    top: edge,
    bottom: edge,
    left: edge,
    right: edge,
    insideHorizontal: edge,
    insideVertical: edge,
  };
}

function cell(value, options = {}) {
  return new TableCell({
    width: options.width
      ? { size: options.width, type: WidthType.PERCENTAGE }
      : undefined,
    margins: {
      top: 100,
      bottom: 100,
      left: 120,
      right: 120,
    },
    children: [
      paragraph(value, {
        alignment: AlignmentType.LEFT,
        after: 0,
        size: 20,
        bold: options.bold,
      }),
    ],
  });
}

function metadataTable(content) {
  const rows = [
    ["Name of the Student:", content.student_name],
    ["PRN:", content.prn],
    ["Title of Practical:", content.practical_title],
    ["Faculty Name:", content.faculty_name],
    ["Sign:", "____________________________"],
  ];

  return new Table({
    width: { size: 100, type: WidthType.PERCENTAGE },
    columnWidths: [35, 65],
    borders: borders(),
    rows: rows.map(
      ([label, value]) =>
        new TableRow({
          cantSplit: true,
          children: [
            cell(label, { width: 35, bold: true }),
            cell(value || "", { width: 65 }),
          ],
        })
    ),
  });
}

function sectionText(value) {
  if (value == null || value === "") {
    return [paragraph("To be completed.", { color: GRAY })];
  }

  if (Array.isArray(value)) {
    return value.flatMap((item, index) => [
      new Paragraph({
        numbering: {
          reference: "report-bullets",
          level: 0,
        },
        spacing: { after: 80 },
        children: [text(item)],
      }),
    ]);
  }

  return String(value)
    .split(/\r?\n/)
    .filter((line) => line.trim())
    .map((line) => paragraph(line, { indent: true }));
}

function screenshotPlaceholder(label) {
  return new Table({
    width: { size: 100, type: WidthType.PERCENTAGE },
    borders: borders("999999", 4),
    rows: [
      new TableRow({
        children: [
          new TableCell({
            margins: {
              top: 240,
              bottom: 240,
              left: 160,
              right: 160,
            },
            children: [
              new Paragraph({
                alignment: AlignmentType.CENTER,
                spacing: { after: 80 },
                children: [
                  text(`[ ${label} ]`, {
                    size: 20,
                    color: GRAY,
                  }),
                ],
              }),
              new Paragraph({
                alignment: AlignmentType.CENTER,
                children: [
                  text(
                    "Insert the actual practical screenshot here.",
                    { size: 18, color: GRAY }
                  ),
                ],
              }),
            ],
          }),
        ],
      }),
    ],
  });
}

export async function generateDocx({
  specPath,
  contentPath,
  outputPath,
}) {
  const jobDir = path.dirname(path.resolve(specPath));

  const spec = JSON.parse(
    await fs.readFile(specPath, "utf8")
  );

  const content = JSON.parse(
    await fs.readFile(path.resolve(contentPath), "utf8")
  );

  const extraction = JSON.parse(
    await fs.readFile(
      path.join(jobDir, "extraction.json"),
      "utf8"
    )
  );

  const pageWidthPt =
    spec.page?.width_pt ?? 595.32;

  const pageHeightPt =
    spec.page?.height_pt ?? 841.92;

  const pageWidthTwips = Math.round(pageWidthPt * 20);
  const pageHeightTwips = Math.round(pageHeightPt * 20);

  const logoInfo = spec.header?.elements?.find(
    (item) => item.kind === "image" && item.asset
  );

  let logoBuffer;

  if (logoInfo?.asset) {
    const logoPath = path.resolve(jobDir, logoInfo.asset);

    try {
      logoBuffer = await fs.readFile(logoPath);
    } catch {
      console.warn("Could not load the extracted college logo.");
    }
  }

  const headerChildren = [];

  if (logoBuffer) {
    headerChildren.push(
      new Paragraph({
        alignment: AlignmentType.CENTER,
        spacing: { after: 80 },
        children: [
          new ImageRun({
            data: logoBuffer,
            type: "png",
            transformation: {
              width: 180,
              height: 58,
            },
          }),
        ],
      })
    );
  }

  headerChildren.push(
    new Paragraph({
      alignment: AlignmentType.CENTER,
      spacing: { after: 40 },
      children: [
        text(
          "School of Computer Science, Engineering and Applications (SCSEA)",
          { size: 20, bold: true }
        ),
      ],
    }),
    new Paragraph({
      alignment: AlignmentType.CENTER,
      spacing: { after: 40 },
      children: [
        text("B.Tech TY (CCSA)", { size: 20, bold: true }),
      ],
    }),
    new Paragraph({
      alignment: AlignmentType.CENTER,
      spacing: { after: 100 },
      children: [
        text("Subject : Cloud Networking & Security", {
          size: 20,
          bold: true,
        }),
      ],
      border: {
        bottom: {
          style: BorderStyle.SINGLE,
          size: 6,
          color: BLUE,
          space: 6,
        },
      },
    })
  );

  const header = new Header({
    children: headerChildren,
  });

  const footer = new Footer({
    children: [
      new Paragraph({
        alignment: AlignmentType.RIGHT,
        children: [
          text(`PRN: ${content.prn || ""}    `, { size: 16 }),
          new TextRun({
            children: ["Page ", PageNumber.CURRENT],
            font: FONT,
            size: 16,
          }),
        ],
      }),
    ],
  });

  const body = [];

  body.push(
    new Paragraph({
      alignment: AlignmentType.CENTER,
      spacing: { before: 100, after: 160 },
      children: [
        text("PRACTICAL REPORT", {
          size: 28,
          bold: true,
          color: BLUE,
        }),
      ],
    }),
    metadataTable(content),
    heading("Aim"),
    ...sectionText(content.aim),
    heading("Objectives"),
    ...sectionText(content.objectives),
    heading("AWS Services Used"),
    ...sectionText(content.aws_services_used),
    heading("Prerequisites"),
    ...sectionText(content.prerequisites),
    heading("Architecture Diagram"),
    ...(content.architecture_diagram &&
    !String(content.architecture_diagram)
      .toLowerCase()
      .startsWith("insert ")
      ? sectionText(content.architecture_diagram)
      : [screenshotPlaceholder("Architecture Diagram")]),
    heading("Step-by-Step Implementation")
  );

  const steps = content.step_by_step_implementation;

  if (Array.isArray(steps) && steps.length) {
    steps.forEach((step, index) => {
      body.push(
        new Paragraph({
          keepNext: true,
          spacing: { before: 100, after: 60 },
          children: [
            text(`Step ${index + 1}: `, {
              size: 22,
              bold: true,
              color: BLUE,
            }),
            text(step),
          ],
        })
      );

      const screenshots = content.screenshots;

      if (Array.isArray(screenshots) && screenshots[index]) {
        body.push(screenshotPlaceholder(screenshots[index]));
      }
    });
  } else {
    body.push(...sectionText(steps));
  }

  body.push(
    heading("Screenshot Documentation"),
    ...sectionText(content.screenshot_documentation),
    heading("Verification / Testing"),
    ...sectionText(content.verification_testing),
    heading("Expected Output"),
    ...sectionText(content.expected_output),
    heading("Result / Conclusion"),
    ...sectionText(content.result_conclusion)
  );

  const doc = new Document({
    creator: "ProjectX Template Builder",
    title: content.practical_title || spec.template_name,
    description:
      "Editable practical report generated from an extracted academic template.",
    numbering: {
      config: [
        {
          reference: "report-bullets",
          levels: [
            {
              level: 0,
              format: "bullet",
              text: "\u2022",
              alignment: AlignmentType.LEFT,
              style: {
                paragraph: {
                  indent: { left: 360, hanging: 180 },
                },
              },
            },
          ],
        },
      ],
    },
    sections: [
      {
        properties: {
          page: {
            size: {
              width: pageWidthTwips,
              height: pageHeightTwips,
            },
            margin: {
              top: 1700,
              bottom: 1100,
              left: 1000,
              right: 1000,
              header: 300,
              footer: 400,
            },
          },
        },
        headers: { default: header },
        footers: { default: footer },
        children: body,
      },
    ],
  });

  const finalPath = path.resolve(outputPath);

  await fs.mkdir(path.dirname(finalPath), {
    recursive: true,
  });

  await fs.writeFile(finalPath, await Packer.toBuffer(doc));

  return {
    outputPath: finalPath,
    mode: "editable DOCX reconstruction",
    editableText: true,
    logoIncluded: Boolean(logoBuffer),
    sectionsRendered: body.length,
    note:
      "Text and tables are editable. Screenshot placeholders are not original screenshots. Exact PDF positioning and page borders require further layout matching.",
  };
}
